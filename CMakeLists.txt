cmake_minimum_required(VERSION 3.20)
project(RVC LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
option(RVC_COVERAGE "Instrument production C for gcov" ON)
add_library(rvc_core STATIC src/rvc.c src/fsm.c)
target_include_directories(rvc_core PUBLIC include)
target_compile_options(rvc_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(RVC_COVERAGE AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
  target_compile_options(rvc_core PRIVATE --coverage -O0 -g)
  target_link_options(rvc_core INTERFACE --coverage)
endif()
add_executable(rvc_cli src/main.c)
target_link_libraries(rvc_cli PRIVATE rvc_core)
include(CTest)
if(BUILD_TESTING)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  add_subdirectory(.tools/googletest-1.15.2 googletest EXCLUDE_FROM_ALL)
  add_executable(rvc_tests tests/controller_test.cpp)
  target_link_libraries(rvc_tests PRIVATE rvc_core GTest::gtest_main)
  add_test(NAME rvc_unit_tests COMMAND rvc_tests --gtest_output=xml:${CMAKE_BINARY_DIR}/unit_results.xml)
endif()
